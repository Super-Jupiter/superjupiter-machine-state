cmake_minimum_required(VERSION 3.16)
project(SuperJupiterMachineStateConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_MACHINE_STATE_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_MACHINE_STATE_SOURCE_DIR}" machine-state)
  if(TARGET superjupiter-machine-state-tests OR TARGET superjupiter-machine-state-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterMachineState 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-machine-state-consumer main.cpp)
target_link_libraries(superjupiter-machine-state-consumer PRIVATE SuperJupiter::MachineState)
